class ZYCTeam {
    constructor() {
        this.version = "2.0.0";
        this.teamName = "ZYC TEAM";
    }

    showWelcome() {
        return `⚡ مرحباً بك في ${this.teamName} ⚡\nالإصدار: ${this.version}`;
    }

    async fetchData() {
        // مثال لجلب بيانات من API
        try {
            const response = await fetch('https://jsonplaceholder.typicode.com/posts/1');
            const data = await response.json();
            console.log("تم جلب البيانات:", data);
            return data;
        } catch (error) {
            console.error("خطأ في جلب البيانات:", error);
        }
    }

    start() {
        console.log(this.showWelcome());
        console.log("تم تشغيل السكربت بنجاح!");
        // يمكن إضافة المزيد من الوظائف هنا
    }
}

// تشغيل السكربت
const zyc = new ZYCTeam();
zyc.start();

// تصدير للاستخدام في الوحدات الأخرى
if (typeof module !== 'undefined' && module.exports) {
    module.exports = ZYCTeam;
}
